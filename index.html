quickstart.py • Zero Dipendenze Esterne Python 3.8+ 
from houdinick import HoudinickClient

# Inizializza il client con la tua chiave RapidAPI (Piano Gratuito o Pro)
client = HoudinickClient(api_key="TUA_CHIAVE_API")

# Estrai le gare d'appalto PNRR aperte in Lombardia sopra i 5 Milioni €
response = client.get_tenders(
    category="pnrr_appalto",
    location="Lombardia",
    value_min=5000000,
    limit=5
)

for t in response["items"]:
    print(f"[{t['payload']['codice_cig']}] {t['title']} — €{t['value_num']:,.2f}")
